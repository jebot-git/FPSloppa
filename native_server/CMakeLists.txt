cmake_minimum_required(VERSION 3.20)
project(cq_standalone LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CQ_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../external-tools/cq-standalone" CACHE PATH "Pinned dependency cache")
foreach(option GENERATE_DEBUG_SYMBOLS INTERPROCEDURAL_OPTIMIZATION USE_AVX USE_AVX2 USE_AVX512 USE_SSE4_1 USE_SSE4_2 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER OVERRIDE_CXX_FLAGS FLOATING_POINT_EXCEPTIONS_ENABLED)
 set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "" FORCE)
add_subdirectory("${CQ_DEPS}/jolt/Build" jolt EXCLUDE_FROM_ALL)
add_subdirectory("${CQ_DEPS}/enet" enet EXCLUDE_FROM_ALL)
add_subdirectory(../tools/authority_core authority EXCLUDE_FROM_ALL)
add_library(cq_json INTERFACE)
target_include_directories(cq_json INTERFACE "${CQ_DEPS}/json/single_include")
find_package(Threads REQUIRED)
# The executable source list is explicit so a Godot adapter cannot enter this target.
add_library(cq_server_runtime STATIC src/snapshot_interest.cpp src/config.cpp src/world_data.cpp src/navigation.cpp src/terrain.cpp src/world.cpp src/assets.cpp src/destruction.cpp src/server.cpp src/admin.cpp src/crypto.cpp vendor/fastlz.c)
target_include_directories(cq_server_runtime PUBLIC src "${CMAKE_CURRENT_SOURCE_DIR}/../addons/fps_native/src" "${CQ_DEPS}/enet/include")
target_link_libraries(cq_server_runtime PUBLIC cq_authority cq_json Jolt enet Threads::Threads)
target_compile_options(cq_server_runtime PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(cq-server src/main.cpp)
target_link_libraries(cq-server PRIVATE cq_server_runtime)
add_executable(cq-server-tests tests/tests.cpp)
# CTest also registers the shared authority suite from the excluded subdirectory.
add_dependencies(cq-server-tests authority-tests)
target_link_libraries(cq-server-tests PRIVATE cq_server_runtime)
enable_testing()
add_test(NAME cq-server-unit COMMAND cq-server-tests)
add_executable(cq-load-client tests/load_client.cpp)
target_link_libraries(cq-load-client PRIVATE cq_server_runtime)
