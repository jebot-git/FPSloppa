cmake_minimum_required(VERSION 3.20)
project(fpsloppa_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(FPS_SERVER "Build without avatar bindings" OFF)
set(GODOTCPP_DEFAULT_API_VERSION "4.7")
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
if(FPS_SERVER)
 set(FPS_PROFILE profile-server.json)
else()
 set(FPS_PROFILE profile.json)
endif()
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/${FPS_PROFILE}" CACHE FILEPATH "" FORCE)
add_subdirectory(${GODOT_CPP_PATH} godot-cpp)
add_library(fpsloppa_native SHARED src/codec.cpp src/client_frame.cpp src/client_presentation.cpp src/client_input.cpp src/network_packing.cpp src/destruction_runtime.cpp src/movement_runtime.cpp src/client_state.cpp src/world_runtime.cpp src/world_terrain.cpp src/backend.cpp src/player_simulation.cpp src/snapshot_preparation.cpp src/authority_cq_snapshot.cpp src/authority_cq.cpp src/authority_squads.cpp src/authority_vehicles.cpp src/authority_ai.cpp src/authority_protocol.cpp src/authority_melee.cpp src/authority_round.cpp src/authority_entities.cpp src/authority_abilities.cpp src/authority_ballistics.cpp src/authority_terrain.cpp src/authority_collision.cpp src/authority_queries.cpp src/authority_utilities.cpp src/authority_gameplay.cpp src/authority_snapshot.cpp src/authority_state.cpp src/authority_vr.cpp src/authority_core.cpp src/authority_world.cpp src/authority.cpp)
if(NOT FPS_SERVER)
 target_sources(fpsloppa_native PRIVATE src/vr_interaction.cpp src/client_services.cpp src/avatar_container.cpp src/client_session.cpp src/remote_interpolation.cpp src/standalone_client.cpp src/register.cpp src/avatar_runtime.cpp src/audio_runtime.cpp src/poses.cpp src/pose_preparation.cpp src/avatar_channels.cpp)
else()
 target_sources(fpsloppa_native PRIVATE src/register_server.cpp)
endif()
find_package(Threads REQUIRED)
target_link_libraries(fpsloppa_native PRIVATE godot-cpp Threads::Threads)
target_compile_options(fpsloppa_native PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3;-ffp-contract=off;-Wall;-Wextra>)
set_target_properties(fpsloppa_native PROPERTIES PREFIX "lib")
