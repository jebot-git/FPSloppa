cmake_minimum_required(VERSION 3.20)
project(fpsloppa_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(FPS_SERVER "Build without avatar bindings" OFF)
set(GODOTCPP_DEFAULT_API_VERSION "4.7")
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
if(FPS_SERVER)
 set(FPS_PROFILE profile-server.json)
else()
 set(FPS_PROFILE profile.json)
endif()
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/${FPS_PROFILE}" CACHE FILEPATH "" FORCE)
add_subdirectory(${GODOT_CPP_PATH} godot-cpp)
add_library(fpsloppa_native SHARED src/projectiles.cpp src/codec.cpp src/network_packing.cpp src/bots.cpp src/bot_steering.cpp src/st_steering.cpp)
if(NOT FPS_SERVER)
 target_sources(fpsloppa_native PRIVATE src/register.cpp src/poses.cpp src/pose_preparation.cpp src/avatar_channels.cpp)
else()
 target_sources(fpsloppa_native PRIVATE src/register_server.cpp)
endif()
target_link_libraries(fpsloppa_native PRIVATE godot-cpp)
target_compile_options(fpsloppa_native PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3;-ffp-contract=off;-Wall;-Wextra>)
set_target_properties(fpsloppa_native PROPERTIES PREFIX "lib")
