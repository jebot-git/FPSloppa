cmake_minimum_required(VERSION 3.20)
project(cq_authority_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(CQ_SANITIZE "Address and undefined behavior sanitizers" OFF)
add_library(cq_authority ../../addons/fps_native/src/authority_ai.cpp ../../addons/fps_native/src/authority_protocol.cpp ../../addons/fps_native/src/authority_melee.cpp ../../addons/fps_native/src/authority_round.cpp ../../addons/fps_native/src/authority_entities.cpp ../../addons/fps_native/src/authority_abilities.cpp ../../addons/fps_native/src/authority_squads.cpp ../../addons/fps_native/src/authority_vehicles.cpp ../../addons/fps_native/src/authority_ballistics.cpp ../../addons/fps_native/src/authority_terrain.cpp ../../addons/fps_native/src/authority_collision.cpp ../../addons/fps_native/src/authority_queries.cpp ../../addons/fps_native/src/authority_utilities.cpp ../../addons/fps_native/src/authority_cq_snapshot.cpp ../../addons/fps_native/src/authority_cq.cpp ../../addons/fps_native/src/authority_gameplay.cpp ../../addons/fps_native/src/authority_snapshot.cpp ../../addons/fps_native/src/authority_state.cpp ../../addons/fps_native/src/authority_core.cpp ../../addons/fps_native/src/authority_vr.cpp)
target_include_directories(cq_authority PUBLIC ../../addons/fps_native/src)
target_compile_options(cq_authority PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(authority-tests squad_tests.cpp vr_port_tests.cpp transport_tests.cpp tests.cpp protocol_tests.cpp entity_tests.cpp gameplay_tests.cpp reference_tests.cpp)
target_link_libraries(authority-tests PRIVATE cq_authority)
if(CQ_SANITIZE)
 target_compile_options(cq_authority PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(cq_authority PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME resident-authority COMMAND authority-tests)
