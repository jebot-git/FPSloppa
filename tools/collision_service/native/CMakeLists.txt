cmake_minimum_required(VERSION 3.20)
project(cq_collision_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_library(cq_collision INTERFACE)
target_include_directories(cq_collision INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(collision_unit tests.cpp)
add_executable(collision_suite suite.cpp)
foreach(target collision_unit collision_suite)
 target_link_libraries(${target} PRIVATE cq_collision Threads::Threads)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endforeach()
# suite.cpp deliberately supplies a malloc-backed operator new for allocation accounting.
target_compile_options(collision_suite PRIVATE -Wno-mismatched-new-delete)
enable_testing()
add_test(NAME collision_unit COMMAND collision_unit)

option(CQ_WITH_JOLT "Build the direct Jolt comparison (test only)" OFF)
set(CQ_JOLT_SOURCE_DIR "" CACHE PATH "Existing Jolt 5.6.0 source checkout")
if(CQ_WITH_JOLT)
 set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
 set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
 set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "" FORCE)
 set(DOUBLE_PRECISION ON CACHE BOOL "" FORCE)
 set(CROSS_PLATFORM_DETERMINISTIC ON CACHE BOOL "" FORCE)
 set(USE_ASSERTS ON CACHE BOOL "" FORCE)
 set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
 set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
 set(ENABLE_OBJECT_STREAM OFF CACHE BOOL "" FORCE)
 set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
 foreach(compute DX12 VK MTL CPU_COMPUTE)
  set(JPH_USE_${compute} OFF CACHE BOOL "" FORCE)
 endforeach()
 if(NOT CQ_JOLT_SOURCE_DIR)
  include(FetchContent)
  FetchContent_Declare(cq_jolt
   GIT_REPOSITORY https://github.com/jrouwe/JoltPhysics.git
   GIT_TAG e77f175595e64cb44218cc9d9d56fc365ad0e36a
   SOURCE_SUBDIR Build)
  FetchContent_MakeAvailable(cq_jolt)
 else()
  add_subdirectory("${CQ_JOLT_SOURCE_DIR}/Build" "${CMAKE_BINARY_DIR}/jolt")
 endif()
 add_executable(collision_jolt_suite jolt_suite.cpp)
 add_executable(collision_jolt_unit jolt_tests.cpp)
 add_executable(collision_jolt_blockers jolt_blockers.cpp)
 add_executable(collision_jolt_map_traversal jolt_map_traversal.cpp)
 foreach(target collision_jolt_suite collision_jolt_unit collision_jolt_blockers collision_jolt_map_traversal)
  target_link_libraries(${target} PRIVATE cq_collision Jolt Threads::Threads)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -Wno-mismatched-new-delete)
 endforeach()
 add_test(NAME collision_jolt_unit COMMAND collision_jolt_unit)
 add_test(NAME collision_jolt_blockers COMMAND collision_jolt_blockers)
endif()
